## 석유 시추
### python
```python
# https://school.programmers.co.kr/learn/courses/30/lessons/250136

from collections import deque

def solution(land):
    width,depth = len(land[0])+2,len(land)+2
    l = [[0]*width for _ in range(depth)]
    for r in range(1,depth-1):
        for c in range(1,width-1):
            l[r][c]=land[r-1][c-1]

    visited = [[False]*width for _ in range(depth)]
    c_count=[0]*width #행에서 얻을 수 있는 석유량
    for i in range(1,depth):
        for j in range(1,width):
            # BFS
            if not visited[i][j] and l[i][j]==1:
                visited[i][j]=True
                count = 0
                queue = deque()
                queue.append((i,j))
                c_min,c_max=j,j
                while queue:
                    r,c = queue.popleft()
                    count+=1
                    if not visited[r][c+1] and l[r][c+1]==1:
                        queue.append((r,c+1))
                        if (c+1>c_max): c_max = c+1
                        visited[r][c+1]=True
                    if not visited[r][c-1] and l[r][c-1]==1:
                        queue.append((r,c-1))
                        if (c-1<c_min): c_min = c-1
                        visited[r][c+1]=True
                    if not visited[r-1][c] and l[r-1][c]==1:
                        queue.append((r-1,c))
                        visited[r-1][c]=True
                    if not visited[r+1][c] and l[r+1][c]==1:
                        queue.append((r+1,c))
                        visited[r+1][c]=True
                # 좌표당 석유량 저장
                for k in range(c_min,c_max+1):
                    c_count[k]+=count
    return max(c_count)
```
- BFS로 연결된 석유 덩이를 카운트해 y좌표마다 저장
- 실패
- 안된 이유가 따로 있지 37번줄의 `c+1`이 `c-1`로 수정되어야 한다
