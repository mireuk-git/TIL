### python
```python
def solution(n,info):
    answer = [-1]
    max_diff = 0

    def dfs(idx,arrows,result):
        nonlocal answer, max_diff
        if idx==10: #dfs 종료조건
            result[10]+=n-arrows #남은 화살 땅에 꼬라박기
            #점수계산
            p_score,l_score = 0,0
            for i in range(11):
                if info[i]<result[i]:
                    l_score+=10-i
                elif info[i]>0: # 명중 수가 같더라도 0보다는 커야 어피치가 득점 
                    p_score+=10-i

            diff = l_score - p_score
            if max_diff<diff: #최고 격차
                max_diff = diff
                answer = result[:]
            elif diff == max_diff and diff>0: #격차가 같다면 낮은 점수부터 비교
                for i in range(10,-1,-1):
                    if result[i] > answer[i]:
                        answer = result[:]
                        break
                    elif result[i]<answer[i]: break #저점 갱신실패
            result[10]=0 #backtracking 원상복구
            return
        # idx점 포기 선택
        dfs(idx+1,arrows,result)
        # idx점 공략 선택
        if arrows+info[idx]+1<=n:
            result[idx] = info[idx]+1
            dfs(idx+1,arrows+info[idx]+1,result)
            result[idx] = 0 #backtracking 원상복구

    dfs(0,0,[0]*11)
    return answer
```
- 백트래킹 `dfs` 사용
