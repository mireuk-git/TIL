```python
def solution(cap,n,deliveries,pickups):
    distance = 0 #총 이동거리
    
    di = n-1 #delivery index
    pi = n-1 #pickup index

    while di>=0 or pi>=0:
        #search counter
        while di >= 0 and deliveries[di]==0: di-=1
        while pi >= 0 and pickups[pi]==0: pi-=1
        furthest = max(di,pi)
        if furthest<0: break
        distance+=(furthest+1)*2
        #delivery
        capacity = cap
        while di>=0 and capacity>0:
            use = min(deliveries[di],capacity)
            deliveries[di] -= use
            capacity -= use
            if deliveries[di]==0: di-=1
        #pickup
        capacity = cap
        while pi>=0 and capacity>0:
            use = min(pickups[pi],capacity)
            pickups[pi]-=use
            capacity-=use
            if pickups[pi]==0: pi-=1
    return distance

deliveries = [1, 0, 3, 1, 2]
pickups = [0, 3, 0, 4, 0]
print(solution(4,5,deliveries,pickups))
```
- greedy: 가장 먼 곳에서부터 배달 시작, 가장 먼 거리를 최소로 다녀올 수 있게 최적화
