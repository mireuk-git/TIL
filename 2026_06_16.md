# Codekata
## 석유 시추
### java
```java

import java.util.LinkedList;
import java.util.Queue;

class Solution{
    int solution(int[][] land){
        int depth = land.length+2;
        int width = land[0].length+2;
        int[][] l = new int[depth][width];
        for (int d=1;d<depth-1;d++){
            for (int w=1;w<width-1;w++){
                l[d][w] = land[d-1][w-1];
            }
        }

        boolean[][] visited = new boolean[depth][width];
        int[] oil = new int[width];
        for (int d=1;d<depth-1;d++){
            for (int w=1;w<width-1;w++){
                if (!visited[d][w] && l[d][w]==1){
                    int count = 0;
                    Queue<int[]> q = new LinkedList<>();
                    q.add(new int[] {d,w});
                    int c_min = w;
                    int c_max = w;

                    while(!q.isEmpty()){
                        int[] pos = q.poll();
                        int r = pos[0], c = pos[1];
                        count++;
                        visited[r][c] = true;
                        if (!visited[r][c+1] && l[r][c+1]==1){
                            q.add(new int[] {r,c+1});
                            if (c_max<c+1) c_max=c+1;
                            visited[r][c+1]=true;
                        }
                        if (!visited[r][c-1] && l[r][c-1]==1){
                            q.add(new int[] {r,c-1});
                            if (c_min>c-1) c_min=c-1;
                            visited[r][c-1]=true;
                        }
                        if (!visited[r+1][c] && l[r+1][c]==1){
                            q.add(new int[] {r+1,c});
                            visited[r+1][c]=true;
                        }
                        if (!visited[r-1][c] && l[r-1][c]==1){
                            q.add(new int[] {r-1,c});
                            visited[r-1][c]=true;
                        }
                    }
                    for (int c=c_min;c<=c_max;c++){
                        oil[c] += count;
                    }
                }
            }
        }

        int highscore = 0;
        for (int s: oil) if (s>highscore) highscore = s;
        return highscore;
    }
}
```
- `queue`를 써서 BFS탐색
