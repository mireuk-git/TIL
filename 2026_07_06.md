def solution(board):
    count = {"O":0,"X":0}
    char = ["O","X"]
    bingo = False

    for i in range(3):
        for j in range(3):
            for c in char:
                if board[i][j] in c:
                    count[c]+=1
                    if board[i]==c*3: 
                        if bingo: return 0
                        bingo = True
                    elif board[0][j]+board[1][j]+board[2][j]==c*3:
                        if bingo: return 0
                        bingo = True
    
    if count["O"] not in [count["X"],count["X"]+1]:
        return 0
            
    return 1

board=["OOO",".O.","..O"]
print(solution(board))
```
- 몇가지 오류가 있어 빠르게 다시 시작한다
- O,X 개수는 각각이 빙고일 때 개수가 정해지게 되며, O가 빙고일 경우 `O=X+1`, X의 경우 `X=O`로 고정되어야 한다
- 같은 행이 여러번 검사되어 생기는 문제

def is_win(board,player):
    for i in range(3):
        if board[i]==player*3:
            return True
        elif board[0][i]+board[1][i]+board[2][i]==player*3:
            return True
    if board[0][0]==board[1][1]==board[2][2]==player:
        return True
    elif board[0][2]==board[1][1]==board[2][0]==player:
        return True
    return False

def solution(board):
    count=[0,0]
    for i in range(3):
        count[0]+=i.count("O")
        count[1]+=i.count("X")
    
    if is_win(board,"O"):
        if count[0]-count[1]!=1: return 0
        else: 
            if is_win(board,"X"): return 0
    elif is_win(board,"X"):
        if count[0]!=count[1]: return 0
    else:
        if count[0]-count[1] not in [0,1]:
            return 0
    return 1
```
