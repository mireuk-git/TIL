### python
```python
from math import gcd,sqrt
def solution(arrayA, arrayB):
    # 1.후보구하기
    #arrayA common divisor
    a=arrayA[0]
    for i in arrayA:
        a=gcd(i,a)
    ad=[a,0] #arrayA common divisor
    for i in range(2,int(sqrt(a))+1):
        if a%i==0: 
            ad.append(i)
            ad.append(a//i)
    ad.sort(reverse=True)

    #arrayB common divisor
    b=arrayB[0]
    for i in arrayB:
        b=gcd(i,b)
    bd=[b,0] #arrayB common divisor
    for i in range(2,int(sqrt(b))+1):
        if b%i==0:
            bd.append(i)
            bd.append(b//i)
    bd.sort(reverse=True)

    # 2.조건을 만족하는 최대값 찾기
    ap,bp=0,0 #pointer
    while True:
        if ad[ap]==bd[bp]==0: return 0 #ZeroDivisionError
        divided=False
        #arrayA common divisor is greater
        if ad[ap]>bd[bp]:
            for b in arrayB:
                #all arrayB % arrayA common divisor should be 0
                if b%ad[ap]==0:
                    divided=True
                    break
            if not divided: return ad[ap] #success; return answer
            ap+=1 #prepare next arrayA common divisor
        #arrayB common divisor is greater
        else:
            for a in arrayA:
                #all arrayA % arrayB common divisor should be 0
                if a%bd[bp]==0:
                    divided=True
                    break
            if not divided: return bd[bp] #success; return answer
            bp+=1 #prepare next arrayB common divisor
```
- 조건을 만족하는 최대값 구하기이기 때문에 먼저 후보군을 추리고 시작한다
- 한쪽 집단의 모든 수를 나눌 수 있어야 하기 때문에 그 집단의 공약수가 답의 후보군이 될 수 있다
- 집단 별로 공약수의 집합(`ad`,`bd`)을 계산, 최대값을 구해야 하기 때문에 내림차순으로 정렬
- 양쪽 후보군 중 큰 순으로 조건을 만족하는 지 검사
- 반대편 집합에서 단 하나라도 나누어 떨어지는 게 있다면 검사를 중단하고 바로 다음 후보 검사
- 내림차순으로 검색하기 때문에, 단 하나도 나누어 떨어지는 게 없다면 정답으로 판정, 리턴

### java
```java
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

class Solution{
    private int gcd(int a, int b){
        while (b>0){
            int tmp = a;
            a = b;
            b = tmp%b;
        }
        return a;
    }
    public int solution(int[] arrayA, int[] arrayB){
        int a = arrayA[0];
        for (int i: arrayA)
            a = gcd(i,a);
        List<Integer> a_divisor = new ArrayList<>();
        for (int i=1;i<=Math.sqrt(a);i++){
            if (a%i==0){
                a_divisor.add(i);
                a_divisor.add(a/i);
            }
        }
        Collections.sort(a_divisor,Collections.reverseOrder());

        int b = arrayB[0];
        for (int i: arrayB)
            b = gcd(i,b);
        List<Integer> b_divisor = new ArrayList<>();
        for (int i=1;i<=Math.sqrt(b);i++){
            if (b%i==0){
                b_divisor.add(i);
                b_divisor.add(b/i);
            }
        }
        Collections.sort(b_divisor,Collections.reverseOrder());

        int a_pointer=0, b_pointer=0;
        while (true){
            if (a_divisor.get(a_pointer)==1 && b_divisor.get(b_pointer)==1)
                return 0;
            boolean divided = false;
            if (a_divisor.get(a_pointer)>b_divisor.get(b_pointer)){
                for (int i: arrayB){
                    if (i%a_divisor.get(a_pointer)==0){
                        divided = true;
                        break;
                    }
                }
                if (!divided) return a_divisor.get(a_pointer);
                a_pointer++;
            }
            else{
                for (int i:arrayA){
                    if (i%b_divisor.get(b_pointer)==0){
                        divided = true;
                        break;
                    }
                }
                if (!divided) return b_divisor.get(b_pointer);
                b_pointer++;
            }
        }
    }
}
```
